Avoid the time wasted each time I tpe a query. Instead of this, the query is stored in a VIEW.

CREATE VIEW myview AS
    SELECT name, temp_lo, temp_hi, prcp, date, location
        FROM weather, cities
        WHERE city = name;

SELECT * FROM myview;

SELECT orders.id, orders.customer_id, orders.total_amount, orders.status AS order_status,payments.payment_method, payments.payment_status, payments.payment_date FROM orders JOIN payments ON orders.id = payments.order_id;
